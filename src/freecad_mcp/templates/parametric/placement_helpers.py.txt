import math as _mcp_math


def _mcp_as_vector(_val, _default=(0.0, 0.0, 0.0)):
    if isinstance(_val, FreeCAD.Vector):
        return _val
    if isinstance(_val, dict):
        return FreeCAD.Vector(
            float(_val.get('x', _default[0])),
            float(_val.get('y', _default[1])),
            float(_val.get('z', _default[2])),
        )
    if isinstance(_val, (list, tuple)) and len(_val) >= 3:
        return FreeCAD.Vector(float(_val[0]), float(_val[1]), float(_val[2]))
    if _val is None:
        return FreeCAD.Vector(*_default)
    raise TypeError('Expected Vector-like value, got %r' % (_val,))


def _mcp_dict_to_rotation(_val):
    if isinstance(_val, FreeCAD.Rotation):
        return _val
    if not isinstance(_val, dict):
        raise TypeError(
            'Rotation value must be a dict or FreeCAD.Rotation, got %r' % (_val,)
        )
    if any(_key in _val for _key in ('Yaw', 'Pitch', 'Roll')):
        return FreeCAD.Rotation(
            float(_val.get('Yaw', 0)),
            float(_val.get('Pitch', 0)),
            float(_val.get('Roll', 0)),
        )
    return FreeCAD.Rotation(
        _mcp_as_vector(_val.get('Axis', {}), (0.0, 0.0, 1.0)),
        float(_val.get('Angle', 0)),
    )


def _mcp_dict_to_placement(_offset):
    """Build a Placement from public JSON; Rotation.Angle is in degrees."""
    if isinstance(_offset, FreeCAD.Placement):
        return _offset
    if not isinstance(_offset, dict):
        raise TypeError(
            'Placement value must be a dict or FreeCAD.Placement, got %r'
            % (_offset,)
        )
    if 'Base' in _offset:
        _pos = _offset['Base']
    elif 'Position' in _offset:
        _pos = _offset['Position']
    else:
        _pos = {}
    _base = _mcp_as_vector(_pos)

    _rot = _offset.get('Rotation', {})
    if isinstance(_rot, FreeCAD.Rotation):
        _rotation = _rot
    elif isinstance(_rot, dict):
        _rotation = _mcp_dict_to_rotation(_rot)
    elif _rot is None:
        _rotation = FreeCAD.Rotation()
    else:
        raise TypeError(
            'Rotation value must be a dict or FreeCAD.Rotation, got %r' % (_rot,)
        )
    return FreeCAD.Placement(_base, _rotation)


def _mcp_placement_to_dict(_pl):
    """Serialize Placement; Rotation.Angle is converted to public degrees."""
    return {
        'Base': {
            'x': float(_pl.Base.x),
            'y': float(_pl.Base.y),
            'z': float(_pl.Base.z),
        },
        'Rotation': {
            'Axis': {
                'x': float(_pl.Rotation.Axis.x),
                'y': float(_pl.Rotation.Axis.y),
                'z': float(_pl.Rotation.Axis.z),
            },
            'Angle': float(_pl.Rotation.Angle) * (180.0 / _mcp_math.pi),
        },
    }
