"""A C901-only negative fixture; it is still one coherent capability."""


def classify_request(value: str | None, enabled: bool, retries: int) -> str:
    if value is None:
        return "missing"
    if not enabled:
        return "disabled"
    if value == "cancel":
        return "cancelled"
    if value == "retry":
        if retries == 0:
            return "retry-first"
        if retries == 1:
            return "retry-second"
        if retries == 2:
            return "retry-third"
        return "retry-exhausted"
    if value == "create":
        return "create"
    if value == "delete":
        return "delete"
    if value == "update":
        return "update"
    if value == "query":
        return "query"
    return "unknown"
